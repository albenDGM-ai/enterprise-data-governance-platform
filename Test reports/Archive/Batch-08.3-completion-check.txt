@router.patch(
    "/transformations/{lineage_transformation_id}",
    response_model=LineageTransformationResponse,
)
def update_transformation(
    lineage_transformation_id: UUID,
    payload: LineageTransformationUpdate,
    x_user: str = Header(..., alias="X-User"),
    service: LineageTransformationService = Depends(get_transformation_service),
) -> LineageTransformationResponse:
    entity = service.get(lineage_transformation_id)

    if entity is None:
        raise HTTPException(
            status_code=status.HTTP_404_NOT_FOUND,
            detail="Lineage Transformation not found.",
        )

    try:
        entity = service.update(
            entity,
            modified_by=x_user,
            sequence_number=payload.sequence_number,
            transformation_name=payload.transformation_name,
            transformation_type=payload.transformation_type,
            description=payload.description,
            expression=payload.expression,
            status=payload.status,
        )
        service.session.commit()
        service.session.refresh(entity)
        return entity
    except ValueError as exc:
        service.session.rollback()
        raise HTTPException(
            status_code=status.HTTP_409_CONFLICT,
            detail=str(exc),
        ) from exc
    except Exception:
        service.session.rollback()
        raise
